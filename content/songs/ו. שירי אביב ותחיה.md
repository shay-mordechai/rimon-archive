# ו. שירי אביב ותחיה

## 🔗 קישורים ומידע ארכיוני

(א.)

אור על הכרמל,

עמי נצח אוֹיְבָיו,

יְחִיפָה הָעִיר בְּיָדָיו. מכפרים עוד ינובו,

כָּל חֵלֶק וְגּוֹרָל בָּאָרֶץ

לנו יהיה.

מִטְבְרְיָה בְּיַד בְּנֵי עַמִי,

כָּל הַגָּלִיל לָנוּי יִהְיֶה

יַעֲלֵךְ לִמְרוֹם הַלְבָנוֹן.

גם הַדָרוֹם יִהְיֶה בְּיָדִי נוי,

וגם הצפון,

רְצוֹ נְנוּ בְּשְׁנֵי עֶבְרֵי הַיַּרְדֵּן

מן הַלְבָנוֹן נַחְלֹשׁ

עַל הָעוֹלָם כולו.

הן שוֹנֵא לָנוּ הָאָדָם,

לֹא יַחְפּוֹץ הַשָׁלוֹם

לֹא יחדַּל מִנַּדוֹתֵינוּ,

וְהַלְבָנוֹן מָשָׁל תּבֵל כֻּלָּהּ.

פָּתַח לְבָנוֹן וְלָתֶיךָ

ותאכל אש בארזיך\...

(ב.)

רננת המלך החסיד,

נְעִים זְמִירוֹת

עַל פִּי.

הָדוּר אַחֲרֶיךָ אֶקְרָא

לְאָבָל גּוֹמֵר עָלַי,

הַצִילֵנִי נָא מֵאוֹיְבִי,

מִשְׁכֵנִי הָרָעִים.

וּמֵעַמִּים הַמִּתְפַּשְׁטִים

עַד לֹא רָחוֹק.

הָיִיתִי לִשְׁחוֹק וְקֶלֶס לָמוֹ,

הכְבִּידוּ. עָלוּ. עָלַי,

דִּכְאוּ. נַפְשִׁי.

אֵין מַעֲצוֹר לְךָ אֱלוֹקִי

לָהוֹשִׁיעַ,

הושיעני מיד רעים.

וְיִשְׁמָעֵאל הֵבִיאוּ

עלי דורות רַבִּים

בַּלָּהוֹת,

חָפצוּ לִגְזֹל עדוּתִי

מִיָּדִי,

לקַלֵּס בְּכָל קִנְיָנִי קָדְשִׁי

ולִלְעוֹג לְנַפְשִׁי יְרוּחִי.

וְגוּפִי נָפַל בִּידֵיהֶם,

מֵתַי לְלֹא עֵת, חִנָּם,

עַתָּה מִקִּבְרִי אֲנִי קָם,

כִּי יָעַרְתָּ לִּי נֶצַח,

הוֹשִׁיעָה אֵל!

(ג.)

אָמַרְתָּ אֵלַי

כִּי בְּכָל הַגּוֹיִים תַּעֲשֶׂה כָלָה,

וְאֶת עַמִּי לֹא תַעֲשֶׂה כָּלָה.

אָמַרְתָּ כִּי קָצַפְתָּ מֵעַל עָלֵינוּ,

יְהֵם עֶזְרִי לְבָעָה,

אָמַרְתָּ אִם יָשׁוּבוּ אֵלֶיךָ שְׁכֵנֵינוּ,

תִּטַּע גַּם אוֹתָם בְּתוֹכֵנוּ.

יָדְךָ יְרִיתֶם כָּל הָאָדָם

כִּמְבַקְשׁוֹ אוֹרְךָ גַּם הוּא.

הָיָה לָנוּ לִנְתִיב אוֹר,

בְּבוֹא עָלֵינוּ אוֹר חִנָּם,

לֹא יִשְׁפְּכוּ דָמֵנוּ,

לֹא יִגְזְלוּ גּוֹרָלֵנוּ,

לֹא יִלְחֲצוּ אָדָם חִנָּם.

לֹא יִפְרְעוּ פְרָעוֹת לְעֵין שֶׁמֶשׁ,

לֹא יַהַרְגוּ אֲנָשִׁים, נָשִׁים וָטַף,

פְּנֵי זְקֵנִים לֹא יָחוֹנוּ.

וּבַחוּרִים מִרְמָס לָמוֹ,

בְּתוּלוֹת לְבֹשֶׁת לָמוֹ,

חַרְבִּי לֹא תוֹשִׁיעֵנִי,

קַשְׁתִּי לֹא יַבְטִיחֵנִי,

בְּךָ אֱלֹקי חֶסֶד

גְּבוּרָתֵנוּ,

היה זְרוֹעַ לָנוּ.

בִפְּגוֹעַ בָּנוּ אוֹיֵב -

נַחֲלָתֵנוּ תִּפְדֶּה סֶּלָה!\...

(ד.)

אֵיךְ בָּזוּ לְצִיּוֹן וְלַעַמָּהּ

שָׁחֲטוּנִי וְהרַגְוּנִי בָּעוֹלָם,

שְׂרָפוּנִי בָּעוֹלָם,

סְקָלוּנִי בָּעוֹלָם,

חֲנָקוּנִי בָּעוֹלָם,

אוֹכל פַּת קֹדֶשׁ

גועה בְּדוֹר אַחֲרוֹן

(ה.)

כִּי זָמַם זֵד יְהִיר עוֹל,

הוּא לֹא מָצָא מָצָא אָוֶן

לִבְכּוֹת צְבִי יִשְׂרָאֵל

אֲשֶׁר עַל בָּמוֹתָיו חֲלָל.

בְּעִנְוּיֵי שְׁאוֹל עִנּוּ אוֹתָם,

בָּא לְעינְךָ הַתּוֹכֵחָה הַכְּתוּבָה

בַּתּוֹרָה,

וַאֲשֶׁר לֹא כָּתוּב בָּהּ,

בּנְפֹל הַלֵּב

הַנֶּפֶשׁ תִּשְׁפַּל,

אֵלֶיךָ יִצְעֲקוּ כָּל הָאָדָם,

חָנֵּנוּ אֱלֹקִים,

כִּי הִבְטַחְתָּ

בְּחָקְרֵנוּ אֵלֶיךָ מִצָּר,

כִּי תִשְׁמַע לָנוּ.

חֲשׁוֹב תְּרַחֵמֶנוּ.,

וְשׁוֹא לֹא נאמר

כִּי אִם יֵשֶׁב וְקִבֵּץ\...
