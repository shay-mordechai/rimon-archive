# א. רבי אברהם גר צדק

## 🔗 קישורים ומידע ארכיוני

לְפָנִים בְּאֶרֶץ רוּסְיָה הָיָה גְרֵף פּוֹטוֹצְקִי,

לוֹ אֲחוּזוֹת רַבּוֹת, וְשִׁמְעוֹ יָצָא לְמֵרָחוֹק.

בֶּן יָחִיד הָיָה לוֹ, סְגֻלָּה בֵּין אֲנָשִׁים,

אָהַב דַּעַת, הִשְׁתַּלֵּם בְּכָל חָכְמָה וּמַדָּע,

וּבִתְכוּנַת נַפְשׁוֹ כָּל יְקָר, פְּאֵר נְדִיבוּת;

וַיֹּאהֲבוּהוּ הוֹרָיו, וְרוֹזְנֵי הָאָרֶץ כִּבְּדוּהוּ:

כַּאֲשֶׁר בָּאֵר אָהַב לָהֶגּוֹת בְּכִתְנֵי-הַחֹדֶשׁ,

נָקְרָה בְּלִבּוֹ מַחֲשָׁבָה בְּלִי הֶרֶף: אֵל אֶחָד

בָּרָא אֶת הָעוֹלָם, וּלְיִשְׂרָאֵל עַם סְגֻלָּה

נָתַן תּוֹרָתוֹ, וְלֹא יָמִיר הָאֵל דָּתוֹ,

וְלָמָּה זֶה הָאֱמֶת נֶעְדֶּרֶת מִמֶּנִּי,

וּכְבֶן אֲצִילִים תְּכַסֵּנִי עֲלָטָה לָעַד.

לֹא עָצַר כֹּחַ, וְאֶת לִבּוֹ גִּלָּה אֶל הוֹדָיו,

כִּי תִשְׁגֶּה בַּהֶבֶל וַיֹּאמֶר לְחַלֵּךְ הוֹד הַגְּרָפִים,

כִּי רוֹצֶה הַגֵּר לְקַבֵּל עָלָיו דַּת יִשְׂרָאֵל;

רַק בָּהּ רֹאִים אֶת הַשֶּׁמֶשׁ, וְהָאוֹר מָתוֹק\...

נָבְהֲלוּ הוֹרָיו וַיֹּאמְרוּ לוֹ: מַה זֶּה לָךְ

וּבִמְדִינָתֵנוּ חוֹק לַאֲשֶׁר יַעֲזֹב דָּתוֹ לְהִשָּׁרֵף,

וְכָל טוּבְךָ יֵלֵךְ מִמְּךָ, כָּל הָאֹשֶׁר הַשָּׁמוּר לָךְ.

רָאָה הַבֵּן כִּי אֵין תִּקְוָה לוֹ מֵהוֹרָיו,

וּבְלִבּוֹ אַהֲבָה לֵאלֹהֵי יִשְׂרָאֵל וּלְדָתוֹ,

וַיָּקָם וַיֵּלֵךְ בְּאִישׁוֹן לַיְלָה וַאֲפֵלָה

אֶל אֲשֶׁר יִשָּׂאוּהוּ רַגְלָיו; עָנִי וְנָכְרִי יָבוֹא

אֶל מוֹשַׁב יְהוּדִים, שָׁם יָבוֹא אֶל חֶבְרָתָם,

יְקַבֵּל דָּתָם, הָעָם עַמּוֹ וֵאלֹהֵיהֶם אֱלֹהָיו.

זָכַר כָּל חֶמְדַּת עֲלוּמָיו, וְאֶת כָּל טֹהַר רוּחוֹ:

לֹא שָׁתָה יַיִן לִשְׂכָּרָה, וְחֶבְרַת נָשִׁים מָאֵס,

כַּחֲלוֹם עָבְרוּ עָלָיו חַיָּיו בְּמַדָּע חָכְמָה

לְרַחֲמֵי אֵל רַק צִפָּה, בְּחֹן נְדוּדִים וְאֶבְיוֹנִים,

עַד כְּאַבְרָהָם נִגְלָה עָלָיו כְּמַנְהִיג הַבִּירָה,

וְעַתָּה הוּא הוֹלֵךְ כְּלִצְוֹוֹ, לִנְחֹל אֶת אֲמִתּוֹ.

הַרְבֵּה חָשַׁב בְּדַרְכּוֹ בֶּן הַגְּרָף פּוֹטוֹצְקִי,

עַד בּוֹאוֹ אֶל וִילְנָה עִיר וְאֵם בְּיִשְׂרָאֵל,

שָׁם הִתְוַדַּע אֶל חֲכָמֶיהָ, אֲנָשִׁים רָמֵי הַמַּעֲלָה,

הֲבִיאוּהוּ תַּחַת בְּרִית אֵל, גַּם נִמּוֹל עַל יָדָם,

וּשְׁמוֹ קָרְאוּ אַבְרָהָם, גַּם כְּנוּהוּ גֵּר צֶדֶק.

אָמְרוּ לוֹ: כְּאַבְרָהָם הָעִבְרִי אוֹר תִּהְיֶה בְּגִזְעוֹ,

יְהוּא גֵּר צֶדֶק נֶגְדּוֹ אֵל בַּחֲצִי הַלַּיְלָה, כֵּן תְּנַצֵּחַ

מַאֲכֶל גָּלוּתֵנוּ, וּלְרַגְלֶךָ יַבְקִיעַ עָלֵינוּ אֵל שַׁחֲרִים.

וּמוֹרֶה וּמְלַמֵּד לְהוֹעִיל נָתְנוּ לוֹ מֻפְלָג בַּתּוֹרָה,

לְלַמְדוֹ תּוּשְׁיָה, כִּי אִישׁ בִּינוֹת הוּא, וְרַב לוֹ יָדוֹ.

יָשַׁב גֵּר צֶדֶק בְּבֵית מוֹרֵהוּ. לָמַד הַרְבֵּה,

וּמְבִינֵי דָבָר יֹאמְרוּ כִּי יִהְיֶה לְגָאוֹן וּלְתִפְאֶרֶת;

תַּחַת חַיֵּי עוֹשֶׁר יִנְחַל חַיֵּי-רוּחַ דִּיוֹ,

וּלְמוֹרֵהוּ בַּת, יָפָה וַחֲמוּדָה אֵין לְשַׁעֵר,

וְשֵׁם לָהּ בְּטֹהַר רִיחָה וּבְצְנִיעוּתָהּ,

וַתֶּאֱהַב אוֹתוֹ אַהֲבָה רַבָּה כְּמִיכַל אֶת דָּוִד.

וַיֹּאמֶר אָבִיךְ בַּעֲצַת טוֹבִים לִתְּנָהּ אֶל חֵיקוֹ,

וְהִנֵּה אוֹי וַאֲבוֹי! לֹא רָחוֹק גָּר שָׁם חַיָּט,

וְהוּא אִישׁ רַע וּבְלִיַּעַל, וְלֹא יָדַע אִישׁ מְהוּתוֹ,

וּלְפֶתַע עָשָׂה רָע וְזָדוֹן, שְׁאוֹל חֵטְא,

בְּגָלוֹתוֹ אָזְנֵי הַמֶּמְשָׁלָה, כִּי בֶן-הַגְּרָף פּוֹטוֹצְקִי,

וִילְנָה נִמְצָא, גַּם קִבֵּל עָלָיו דַּת יִשְׂרָאֵל;

וּמַהֵר הוּבָא גֵּר צֶדֶק לִפְלִילִים אֶל עִיר הַבִּירָה.

תְּלַלְת הוֹד הַגְּרָפִים, אָמְרוּ לוֹ הַשּׁוֹפְטִים,

נָאַצְתָּ שֵׁם מְשִׁיחֵנוּ, בְּקַבֶּלְךָ עָלֶיךָ דָּת אַחֶרֶת,

הִשְׁלַכְתָּ לְעָפָר גֶּזֶר הַיִּרְאִים בַּעֲבוּר עַם סְפִין

וְהֵמִיר דָּתֵנוּ בְּחֹקְנוּ, עָלֶיךָ לְהִשָּׁרֵף בְּאֵשׁ.

פִּתְאֹם נָחָה הָרוּחַ עַל גֵּיא אֶרֶץ, וַיְדַבֵּר רָמוֹת:

בְּהִתְנַדְּבִי לַעֲשׂוֹת חֶסֶד בְּבֵית אֲבִי עִם עָנִי וְאֶבְיוֹן,

גַּם בְּחָקְרִי כִּתְבֵי הַחֹדֶשׁ, פַּעֲמַתְנִי רוּחַ אֱלֹהִים,

הִיא אָמְרָה לִּי הַפֵּר בּוֹרְאֲךָ לְאַבְרָהָם הָעִבְרִי,

הוּא אֶחָד וְאֵין שֵׁנִי לוֹ לְהַמְדִּיל לְהַחֲנִיק רֹאשׁ,

לְעוֹלָם יֵשׁ לְעַמּוֹ מְקוֹם הַתּוֹרָה לְעֵינֵי כֹל,

לְעֵינֵי הַשֶּׁמֶשׁ יֵלְכוּ וְיִנְצֹחוּ, וְלֹא יִהְיוּ לְעַבְדוּת,

עַל גּוֹי עֲבָדִים יָנוּחוּ וְיִתְנַשְּׂאוּ, לְהַמְלִיךְ הַתּוֹרָה

עַל הַלְּבָבוֹת, וְלֹא יִהְיוּ עוֹד חֲלוּקִים לִשְׁנֵי עַמִּים,

אֲחוּתָהּ בְּעַמִּים תִּמְלוֹךְ, וְהַמַּמְלָכוֹת כֻּלָּם

אֵלֶיהָ יִפְנוּ עַל פִּיהֶם יְדַבֵּרוּ.

כְּשָׁמְעוּ הַשּׁוֹפְטִים אֶת דְּבָרָיו, צִוְּוּ לַעֲשׂוֹת מְדוּרַת-אֵשׁ,

גַּם הִשְׁלִיכוּהוּ אֶל תּוֹכָהּ. פָּשַׁט הוּא מְמַלֵּא פִּיו:

\"שְׁמַע יִשְׂרָאֵל ה\' אֱלֹהֵינוּ ה\' אֶחָד\", ונשרף.

שָׁמְעוּ בָּחוּרֵי זִילְנָה, רַבִּי תּוּשְׁיָה וְדַעַת,

כִּי יִשְׂבוּ אֵשׁ אֶרֶץ, וְאַפְרוֹ טָמוּן בְּעִיר הַמַּלְכוּת,

קָמוּ בְּאִישׁוֹן לַיְלָה וַיִּפְלָחוּ, וַיֵּלְכוּ לָאוֹר בְּרָקִים וְסַעַר

עַד פּוֹאָם אֶל עִיר הַבִּירָה, גַּם סָכְנוּ נַפְשָׁם

וַיּוֹצִיאוּ אֶפְרוֹ מִקִּבְרוֹ, וַיְבִיאוּהוּ לְוִילְנָה.

טָמְנוּ אֶת אֶפְרוֹ בְּבֵית מוֹעֵד לְכָל חַי שָׁם,

יְהִי צֶמַח וַיֵּצֵא אִילָן עַל קִבְרוֹ בְּדְמוּת אָדָם,

וְתַיָּרִים בָּאִים מֵרָחוֹק לִרְאוֹת אֶת הָאִילָן.
