# הַלְּבָנָה הַמֵּתָה

## 🔗 קישורים ומידע ארכיוני

הִשְׁבַּעְתִּיכֶם, אֵיתָנֵי עוֹלָם:

הָרִים, יָמִּים וּמִדְבָּרִיּוֹת --

אַל נִיד! אַל זִיעַ!

תִּכָּפֵל נָא הַדְּמָמָה שִׁבְעָתַיִם!

עֵרָה וְלֹא עֵרָה נַפְשִׁי מִן חֲלוֹמָהּ,

וְאֵת חֲלוֹמִי הֵן יְדַעְתֶּם:

גַּן-קְבָרִים שָׁבוּי רִשְׁתוֹת-זָהָב\...

שָׁם שֶׁמֶשׁ-כַּדְכֹּד תָּהֵל אוֹרָהּ,

שַׁפְרִיר-שְׁחָקִים יַרְעִיף כַּסְפּוֹ,

שְׂדוֹת-חָרוּץ יַזִּילוּ הוֹדָם,

מַעְיַן-כֶּתֶם יַרְנִין מֵימָיו,

עַל שְׂפָתָהּ שׁוֹמְרָהּ יַעֲלַת-חֵן

פְּרוּעַת שְׁעַר-זְהָבָהּ.

אֵלֶיהָ עֵינַי אִם אָרִימָה,

אֵל שֵׁישׁ-חָזָה בְּקֶסֶם אֶסָּחֵף,

בְּחֵיקָהּ עַרְשִׂי לִי תַּצִּיעַ

רֵיחַ קֵיצָהּ יַשְׁכִּירֵנִי,

רֶטֶט אֹשֶׁר נַפְשִׁי יַעֲבוֹר,

אַחֲרוֹן אוֹנִי בְּמֹתֶק יִכְלֶה,

וְאָז יִרְעַם צְחוֹקָהּ הַפָּרוּעַ

אֱלֵי אָזְנִי:

הַמָּוֶת אָנֹכִי!

עֵרָה וְלֹא עֵרָה נַפְשִׁי מִן חֲלוֹמָהּ

וְאֶת חֲלוֹמִי הֵן יְדַעְתֶּם\...

וָאֶנְעַץ שְׁתֵּי עֵינַי בְּחַלּוֹנִי,

וְעַל הַיָּשִׁישׁ רַב-הַפֹּארוֹת נָפְלוּ:

אִילָן נָאֶה -- כֻּלּוֹ רֹךְ, זֹהַר וְשִׁירָה.

עָלִים לוֹ -- כַּנְפֵי מֶשִׁי צְעִירוֹת

כְּשִׁירַת לְבָבִי קַלִּים וַעֲדִינִים;

אַךְ אָרוּר פּוֹצֶה פִּיו סַפֵּר הוֹדוֹ!

הַיְסֻפַּר זִיו-שְׁקִיעָה עֲלֵי יָמִים,

שִׁירַת-שְׁחָקִים עֶרֶב עַל הַכַּרְמֶל?!

וַאֲנִי -- לִבִּי אוֹמֵר לִי

מַה כָּכָה יִיף הַיָּשִׁישׁ:

בַּת מֶלֶךְ מָדַי הַיְּפֵהפִיָּה

עָיְפָה נָפְשָׁה בַּקֵּשׁ חֲתָנָהּ, -

בְּיַעֲרוֹת עַד בַּמֶּרְחַקִּים

גַּלְמוּד, נֶעֱזָב תּוֹעֶה הוּא.

הַמְּעֻנָּגָה! כַּפּוֹת רַגְלֶיהָ דָּרְכוּ

עַל כָּל קְצוֹת תֵּבֵל מַחֲבוֹאֶיהָ,

אַךְ שֶׁמֶשׁ מִזְרָח, חוֹלוֹ הַצּוֹרֵב

וְשֶׁפַע הוֹדוֹ פֶּרֶא הוֹגִיעוּהָ;

גָּדַל, גָּדַל הַיֵּאוּשׁ בִּלְבָבָהּ,

וְצַעַר רַב אוֹנִים פֹּה כִּי תְּקָפָהּ --

בְּצִפָּרְנֶיהָ, מְגֻדְּלוֹת-פֶּרֶא מִנְּדוֹד,

עֵינֶיהָ -- מַרְגָּלִיּוֹת נִקֵּרָה

וּרְסִיסִים עַל הַיָּשִׁישׁ פִּזְּרָתַם,

לֹא הַרְחֵק אֵל הַיַּרְדֵּן הִיא יוֹשֶׁבֶת,

יְפִי גַּלָּיו אֵינָהּ רוֹאָה -- הֵן עִוְּרָה הִיא,

אַךְ תָּמִיד שִׁירָתָם הִיא מַקְשֶּׁבֶת,

וְעֵת גַּעְגּוּעִים לַחֲתָנָהּ יִתְקְפוּהָ,

חָזֹר תַּחֲזוֹר עַל כָּל אַגָּדוֹתָיו,

טֶרֶם לֶכְתּוֹ סִפֵּר לָהּ.

אֶת מַבָּטִי גָּנְבוּ שַׁחֲקֵי אֱלֹהִים: (במחזה: השיר מתחיל מכאן)

רִנָּה מִתְאַפְּקָהּ, רֵיחַ-שׁוֹשַׁנִּים

וְיָד קְדוֹשָׁה, טְמִירָה טָסָה וּמְפַזְּרָהּ

כּוֹכָבִים -- רְסִיסֵי כֶּתֶם-אוֹפִיר;

לִכְבוֹדוֹ שֶׁל מִי? הִנֵּה הִיא בָּאָה --

כְּלִילַת הַחִוָּרוֹן שֶׁל אוֹצַר שַׁדַּי,

עֲנֻגָּה, כִּצְחוֹק שַׁאֲנָן שֶׁל יֶלֶד תָּמִים,

בָּרָה, כַּשְּׁכִינָה טֶרֶם יָדְעָה יָדְעָה אֵבֶל,

הוֹד גַּעְגּוּעֵי הַנֵּצַח עַל פָּנֶיהָ --

הִנֵּה בָּאָה, הִיא -- הַלְּבָנָה הַמֵּתָה.

שׁוֹכְבָה הִיא בְּקִבְרָהּ, סוּג פִּרְחֵי-תְּכֵלֶת,

סָבִיב לוֹ הֵיכַל-גָּבִישׁ יַבְרִיק הוֹד,

שָׂם בְּקִבְרָהּ גְּוִיַּת-מֶשְׁיָהּ מְסֻתֶּרֶת,

אַךְ רֹאשָׁהּ, אֲבִיב-נְגֹהוֹת, צוֹפֶה פְּנֵי תֵּבֵל;

שַׂעֲרוֹתֶיהָ-פָּז פְּזוּרוֹת עֲלֵי הֵיכָלָהּ;

עֵינַיִם לָהּ -- שְׁנֵי אֶרְאֶלֵּי-אַהֲבָה

הוֹגֵי תְּפִילָּה חֲשָׁאִית, תְּפִילַּת אֵין-סוֹף;

שְׂפָתַיִם לָהּ -- נְשִׁיקַת יָהּ-בָּהּ זָכָּה

מִנִּשְׁמוֹת הָאָדָם לְנָווֹ יָשִׁיב;

הֲבַת אֱלֹהִים הִיא מְאֹד רַכָּה וַעֲנֻוגָּה,

צְמֵאַת מֶרְחַקִּים בֵּית אָבִיהָ עָזְבָה,

וּבַקֵּשׁ יוֹתֵר יֹפִי, פֶּלִי וְנִשְׂגָּב

שְׁבוּיַת-רַעְיוֹן הִיא יָצְאָה לִנְדוּדֶיהָ?

הֲקֶרַע מִגַּעְגּוּעֵי נַפְשׁוֹ הִיא,

סַעֲרַת-אֵלִים גְּדוֹלָה בּוֹ כִּי הִתְחוֹלָלָה,

צָר הָיָה לָהּ מְקוֹמָהּ, בְּחֵמָה נִתְּזָה,

וּכְנָפְשׁוֹ גַם הִיא מְנוּחָה לֹא יָדְעָה?

כְּשִׁירַת יַמִּים מִשְׁתַּפְּכָה אוֹרָתָהּ,

בַּחֲשַׁאי שׁוֹפְעִים נַחֲלֵי-כַּסְפָּהּ;

כְּמַחֳשֶׁבֶת הַיְּצִירָה בְּרֹאשׁ הַיּוֹצֵר,

עֵת בַּתֹּהוּ רִחֲפָה רוּחוֹ עַל פְּנֵי מַיִם;

אוֹקְיָנוֹס קְסָמִים חֲוַרְוָרִים וַעֲגוּמִים

נוֹזֵל פְּלָגִים פְּלָגִים עַל רַחֲבֵי תֵּבֵל,

וְלֹא יֵדַע מַה תְּחִלָּתוֹ וּמַה סוֹפוֹ;

יָם דִּי-נוּר מִשְׂתָּרֵעַ לְבֶן-פָּנִים,

מַמְטִיר יִפְעָהּ, מַחֲרִיד הַיְּקוּם בְּהוֹדוֹ;

כְּשֶׁמֶשׁ אֱלֹהִים בְּיוֹם רִאשׁוֹן לִיצִירָה\...

לְבָנָה יָפָה, רַכָּה וַעֲנֻגָּה!

לִבָּהּ הֶעָדִין הִצְטַעֵר וְחָלָה,

נְפָשׁוֹת כָּלוֹת מִגַּעְגּוּעִים וְאַהֲבָה

צִלֵּי צַעֲרָן הִשְׁרוּ עָלֶיהָ,

מַשָּׂא יְגוֹנָן הֵעִיקוּ עַל נַפְשָׁהּ,

וְכִפְנֵיהֶם הַדַּלִּים חָוְרָה וְעָגְמָה.

הִיא מֵתָה, וְגַם בְּמוֹתָהּ מְנוּחָה לֹא תֵּדָע.

כִּמְקֻלֶּלֶת-אֱלֹהִים הִיא נוֹדֶדֶת,

אוֹר גַּעְגּוּעִים וְאַהֲבָה תְּפַזֵּר בְּבוֹאָהּ

וְצִלְלֵי שְׁחוֹר, מָגוֹר וְיֵאוּשׁ -- בְּצֵאתָהּ; (בשירים: כאן יש רווח)

שְׁנַת הַנֵּצַח תִּישַׁן בְּעֵינַיִם פְּקוּחוֹת,

וְהַכֹּל בַּחֲשַׁאי עוֹד יַרְגִּישׁ לִבָּהּ.

הִנֵּה אֵלַי תִּרְזוֹם וְאִלְּמָה תְּדוֹבֵב:

\"יָדַעְתִּי, עֶלֶם חִוֵּר, גֹּדֶל עֱנוּתְךָ,

יַלְדָּהּ תְּמִימָה תִּנְהַג לִבְּךָ שֶׁבִי,

כְּשֶׁמֶשׁ מַקְדִּים אֵלֶיךָ הִשְׁקִיפָה,

שְׁחוֹרוֹת וַעֲמֻקּוֹת עֵינֶיהָ כַּלַּיִל

וְכַיּוֹם צַחִים וַעֲדִינִים פָּנֶיהָ,

כַּשְּׁקִיעָה -- יַמִּים, יַעַטְפוּךָ קַרְנֶיהָ,

עוֹד נַפְשְׁךָ רוֹעֲדָהּ בִּתְחִנָּה וּתְפִילָּה.

אַךְ בְּצִנְעָה כְּבָר הַיֵּאוּשׁ מַכֶּה גַּלָּיו,

הִנֵּה בְּגַעְגּוּעִים אַתָּה גֹּוֵעַ,

וּצְלָלִים שְׁחוֹרִים עֲלֵי קִבְרְךָ יְרַצְּדוּן.

אֶרְחָמְךָ, אָחִי, צַר לִי בִּגְלָלְךָ,

בְּכַנְפֵי זֹהַר אֲשַׁחֶרְךָ מִשָּׁמַי

וְאֵלַי עֲלֵה, כְּסַפִּיר בְּאוֹרִי זְרַח,

מֶרְחֲבֵי מַעְלָה יְנַחֲמוּךָ בְּעֶדְנָם,

וְצַעַרְךָ יִשְׁקַע בַּנְּדוּדִים\".

הִנֵּה בִּנְגֹהוֹת רַכִּים תִּטְבְּלֵנִי,

דִּמְעָה זַכָּה, טְהוֹרָה תִּזַּל עֵינִי,

וְעַל כַּנְפֵי זֹהַר קַל אֶתְרוֹמָם (במחזה: השיר נגמר כאן)

לָעַלְמָה הַחִוְרָה --

לַלְּבָנָה הַמֵּתָה.

תרע\"ב

זו גרסת כתרים. גרסת במחזה קצרה יותר, מצוין על הטקסט.
